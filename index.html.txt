  <script>
    function yesClicked() {
      document.body.innerHTML = `
        <div class="celebrate">
          <div class="message">
            <h1>YAYYY!!! 💞💞💞</h1>
            <p>You just made me the happiest person ever 🥰</p>
            <div class="icons">🌹🍫💖✨</div>
            <p class="final-line">See you on Valentine’s Day ❤️</p>
          </div>
        </div>
      `;

      const audio = new Audio(
        "https://assets.mixkit.co/sfx/preview/mixkit-soft-pop-bubble-2692.mp3"
      );
      audio.play();

      const symbols = ['💖', '🌸', '✨'];
      for (let i = 0; i < 40; i++) {
        const confetti = document.createElement('div');
        confetti.className = 'confetti-heart';
        confetti.innerText = symbols[Math.floor(Math.random() * symbols.length)];
        confetti.style.left = Math.random() * 100 + 'vw';
        confetti.style.animationDuration = 3 + Math.random() * 3 + 's';
        document.body.appendChild(confetti);
        setTimeout(() => confetti.remove(), 6000);
      }
    }

    function noClicked() {
      const noBtn = document.querySelector('.no');
      const container = document.querySelector('.buttons');

      const maxX = container.clientWidth - noBtn.offsetWidth;
      const maxY = container.clientHeight - noBtn.offsetHeight;

      noBtn.style.left = Math.random() * maxX + 'px';
      noBtn.style.top = Math.random() * maxY + 'px';
    }
  </script>
